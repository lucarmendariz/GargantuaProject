CREATE TABLE product (
    product_id INT AUTO_INCREMENT PRIMARY KEY,
    sku VARCHAR(50) UNIQUE NOT NULL,
    name VARCHAR(100) NOT NULL,
    description TEXT DEFAULT NULL,
    is_active TINYINT(1) DEFAULT 1 NOT NULL,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP
) ENGINE=InnoDB;

CREATE TABLE product_lot (
    lot_id INT AUTO_INCREMENT PRIMARY KEY,
    product_id INT NOT NULL,
    lot_number VARCHAR(100) UNIQUE NOT NULL,
    quantity DECIMAL(12,4) NOT NULL DEFAULT 0.0000,
    measure_unit VARCHAR(20) NOT NULL DEFAULT 'UNITS',
    expiration_date DATE DEFAULT NULL,
    supplier_reference VARCHAR(100) DEFAULT NULL,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    CONSTRAINT fk_lot_product FOREIGN KEY (product_id) REFERENCES product (product_id)
) ENGINE=InnoDB;

CREATE TABLE lot_history (
    history_id INT AUTO_INCREMENT PRIMARY KEY,
    lot_id INT NOT NULL,
    action_type VARCHAR(50) NOT NULL,
    operator VARCHAR(100) NOT NULL,
    action_timestamp TIMESTAMP DEFAULT CURRENT_TIMESTAMP NOT NULL,
    quantity_affected DECIMAL(12,4) DEFAULT NULL,
    ocr_raw_text TEXT DEFAULT NULL,
    observations TEXT DEFAULT NULL,
    CONSTRAINT fk_history_lot FOREIGN KEY (lot_id) REFERENCES product_lot (lot_id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB;

DELIMITER ;;
CREATE OR REPLACE PROCEDURE `ProductLot_GetCurrentStatus`(IN _lotNumber VARCHAR(100))
BEGIN
    IF (SELECT lot_id FROM product_lot WHERE lot_number = _lotNumber) IS NULL THEN
        SIGNAL SQLSTATE '45000' SET MESSAGE_TEXT = 'El lote no existe.', MYSQL_ERRNO = 101;
    END IF;

    SELECT p.product_id, p.sku AS product_sku, p.name AS product_name, pl.lot_id, pl.lot_number, pl.quantity AS initial_quantity, pl.measure_unit, pl.expiration_date, lh.action_type AS current_state, lh.action_timestamp AS last_update, lh.operator AS last_operator, lh.ocr_raw_text, lh.observations AS last_observations
    FROM product_lot pl JOIN product p ON pl.product_id = p.product_id JOIN lot_history lh ON lh.lot_id = pl.lot_id
    WHERE pl.lot_number = _lotNumber ORDER BY lh.history_id DESC LIMIT 1;
END ;;
DELIMITER ;
